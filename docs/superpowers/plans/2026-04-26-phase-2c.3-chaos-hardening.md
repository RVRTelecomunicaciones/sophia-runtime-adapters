# Phase 2C.3 — Chaos + Minimal Hardening — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Land Phase 2C.3 — chaos injection framework, six CI fault scenarios, three safety-net invariant tests, receiver-stub compose overlay, per-PR E2E canary, and nightly comprehensive workflow — followed by docs/ADRs/CHANGELOG and tag `v0.4.0`. Reactive fix bundles open only if chaos surfaces real gaps.

**Architecture:** A single `ChaosAdapter` decorator wraps each `outbound.Adapter` opt-in via env (`RUNTIME_CHAOS_ENABLED=true`, fail-closed in production). A `ChaosReceiptStore` wraps `ports.ReceiptStore` for persist-failure injection. Profiles are validated YAML files; allowlisted; loaded once at bootstrap. Tests range from in-process classification checks (per-PR fast) to a full pipeline canary against `runtime → otel-collector → prometheus → sloth-rendered rules → alertmanager → receiver-stub` in compose. Hardening is chaos-driven: only fixes, no speculation.

**Tech Stack:** Go 1.26, OTel SDK + Prometheus exporter, Sloth, Alertmanager, Prometheus, Docker Compose, `go-yaml/v3`, `testing/quick`, GHA, `gh` CLI.

```go
// Package chaos implements opt-in fault injection for the runtime adapters.
// Chaos-capable code is compiled into the runtime binary but is inert by
// default and fail-closed (R17). Concrete decorators wrap the outbound.Adapter
// port and ports.ReceiptStore; profiles are validated YAML files loaded from
// an allowlist at bootstrap.
//
// See docs/superpowers/specs/2026-04-26-phase-2c.3-chaos-hardening-design.md.
package chaos

import (
	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/valueobjects"
)

// FaultKind enumerates the closed set of injectable fault kinds (spec §5.1,
// D2C3.11). Profile validation rejects unknown kinds at startup.
type FaultKind string

// Timing controls when the fault is applied relative to the wrapped call.
type Timing string

const (
	TimingPreDispatch  Timing = "pre_dispatch"
	TimingPostDispatch Timing = "post_dispatch"
)

// FaultConfig describes a single per-capability fault.
type FaultConfig struct {
	Kind   FaultKind
	Timing Timing
	// Match constrains the fault to a subset of requests; empty = always match.
	Match map[string]string
}

// PersistFaultConfig describes a persist-layer fault (D2C3.8 ChaosReceiptStore).
type PersistFaultConfig struct {
	Kind FaultKind
}

// ExpectedOutcome is the metadata used by integration tests; not consumed by
// the runtime itself (spec §6.1, D2C3.10/D2C3.24).
type ExpectedOutcome struct {
	Status     string
	ErrorClass string
	RetryHint  string
}

// Profile is the loaded, validated representation of a v1 profile YAML.
type Profile struct {
	Version     int
	Name        string
	Description string
	Adapters    map[valueobjects.CapabilityID]FaultConfig
	Persist     *PersistFaultConfig
	Expected    ExpectedOutcome
}

// For returns the fault for the given capability if any.
func (p *Profile) For(c valueobjects.CapabilityID) (FaultConfig, bool) {
	if p == nil {
		return FaultConfig{}, false
	}
	f, ok := p.Adapters[c]
	return f, ok
}

// PersistFault returns the configured persist fault, if any.
func (p *Profile) PersistFault() *PersistFaultConfig {
	if p == nil {
		return nil
	}
	return p.Persist
}

// Config is the bootstrap-resolved chaos configuration.
type Config struct {
	Enabled bool
	Profile *Profile
	Env     string
}
```

```go
	FaultConnectionReset   FaultKind = "connection_reset"
	FaultHangUntilCancel   FaultKind = "hang_until_cancel"
	FaultEIO               FaultKind = "eio"

	FaultSlowBody          FaultKind = "slow_body"
	FaultRedirectLoop      FaultKind = "redirect_loop"
	FaultTLSHandshakeFail  FaultKind = "tls_handshake_fail"
	FaultPersistFailure    FaultKind = "persist_failure"
)

var allFaultKinds = []FaultKind{
	FaultLatency, FaultConnectionReset, FaultHangUntilCancel,
	FaultEIO, FaultENOSPC, FaultPermissionDenied,
	FaultRemoteUnreachable, FaultAuthFailure,
	FaultInjectPanic,
	FaultProcessSignal, FaultProcessExit,
	FaultSlowBody, FaultRedirectLoop, FaultTLSHandshakeFail,
	FaultPersistFailure,
}

// AllFaultKinds returns the closed set of injectable fault kinds.
func AllFaultKinds() []FaultKind {
	out := make([]FaultKind, len(allFaultKinds))
	copy(out, allFaultKinds)
	return out
}

// IsValidFaultKind reports whether k is in the closed enum.
func IsValidFaultKind(k FaultKind) bool {
	for _, candidate := range allFaultKinds {
		if candidate == k {
			return true
		}
	}
	return false
}
```

```go
package chaos

import (
	"context"

	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/services"
	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/valueobjects"
	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/shared"
	"github.com/sophia-ecosystem/runtime-adapters/internal/ports/outbound"
)

// ChaosAdapter wraps an outbound.Adapter and injects faults defined in
// the immutable Profile loaded at bootstrap. R5 holds: chaos returns
// the same AdapterRawOutcome shape as the real adapter would for the
// equivalent real fault (I24).
type ChaosAdapter struct {
	real    outbound.Adapter
	profile *Profile
	clock   shared.Clock
}

// NewChaosAdapter constructs the decorator.
func NewChaosAdapter(real outbound.Adapter, profile *Profile, clk shared.Clock) *ChaosAdapter {
	return &ChaosAdapter{real: real, profile: profile, clock: clk}
}

func (c *ChaosAdapter) ID() valueobjects.AdapterID { return c.real.ID() }

func (c *ChaosAdapter) Capabilities() []valueobjects.Capability {
	return c.real.Capabilities()
}

func (c *ChaosAdapter) Execute(
	ctx context.Context,
	cap valueobjects.Capability,
	payload valueobjects.Payload,
) (services.AdapterRawOutcome, error) {
	fault, ok := c.profile.For(cap.ID())
	if !ok {
		return c.real.Execute(ctx, cap, payload)
	}
	return c.dispatchFault(ctx, cap, payload, fault)
}

// dispatchFault is implemented in Task 1.4 (closed-enum switch).
// For Task 1.3 we provide a stub that delegates to real — establishing
// the structure without any fault behaviour yet.
func (c *ChaosAdapter) dispatchFault(
	ctx context.Context,
	cap valueobjects.Capability,
	payload valueobjects.Payload,
	fault FaultConfig,
) (services.AdapterRawOutcome, error) {
	// Stubbed in 1.3; full switch comes in 1.4.
	return c.real.Execute(ctx, cap, payload)
}
```

### Task 1.4: Fault dispatch — split into 1.4a–1.4f after B1 architectural pivot

> **Pivot recorded post-spec** (2026-04-26, after Tasks 1.1–1.3 landed). See spec Appendix D for the full rationale. TL;DR: Phase 1's per-adapter raw types (`execRaw`, `statusRaw`, `httpRaw`, etc.) are deliberately unexported per R5. The chaos package cannot construct them directly. Resolution: opt-in `ChaosCapable` interface per adapter. Each adapter package owns its own synthesizer for the faults it supports.
>
> Engram entry: `mem_search` topic `phase-2c.3/architectural-pivot-chaoscapable`.
>
> Task 1.4 is now six sub-tasks; old single-task content is superseded. The sub-tasks below use the same TDD pattern as 1.1–1.3.

**Faults split:**

- **Decorator-native** (handled directly in `ChaosAdapter`, no adapter cooperation): `latency`, `hang_until_cancel`, `inject_panic`.
- **Adapter-synthetic** (handled via `ChaosCapable` delegation): `connection_reset`, `eio`, `enospc`, `permission_denied`, `remote_unreachable`, `auth_failure`, `process_signal`, `process_exit`, `slow_body`, `redirect_loop`, `tls_handshake_fail`.
- **Out of both**: `persist_failure` is the `ChaosReceiptStore` wrapper's domain (Task 1.5, unchanged).

```go
package chaos

import (
    "context"

    "github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/services"
    "github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/valueobjects"
    "github.com/sophia-ecosystem/runtime-adapters/internal/ports/outbound"

// ChaosCapable is the opt-in interface implemented by Phase 1 adapters that
// support adapter-synthetic chaos faults. The interface keeps Phase 1's raw
// outcome types unexported (R5) — synthesis happens inside the adapter
// package, using the same private type as a real-fault outcome.
//
// Adapters that do not implement ChaosCapable can still be targeted by
// decorator-native faults (latency, hang_until_cancel, inject_panic) which
// the ChaosAdapter handles without any adapter cooperation.
//
// See spec Appendix D and engram topic phase-2c.3/architectural-pivot-chaoscapable.
type ChaosCapable interface {
    outbound.Adapter

    // SupportedChaosFaults returns the chaos fault kinds this adapter can
    // synthesize for the given capability. The list does NOT include
    // decorator-native faults — those are universally available.
    SupportedChaosFaults(cap valueobjects.Capability) []FaultKind

    // SyntheticOutcome constructs an AdapterRawOutcome of the same shape
    // the adapter would produce for the equivalent real fault. Returns
    // (outcome, true) on success; (nil, false) if the fault is unsupported
    // for this capability (defensive — ValidateProfileSupport should have
    // rejected this case at startup).
    SyntheticOutcome(
        ctx context.Context,
        cap valueobjects.Capability,
        payload valueobjects.Payload,
        fault FaultConfig,
    ) (services.AdapterRawOutcome, bool)
}

// IsDecoratorNative reports whether the fault is handled directly by
// ChaosAdapter without delegating to ChaosCapable.
func IsDecoratorNative(k FaultKind) bool {
    switch k {
    case FaultLatency, FaultHangUntilCancel, FaultInjectPanic:
        return true
    default:
        return false
    }
}
```

```go
// validate_support.go
package chaos

import (
    "errors"
    "fmt"

    "github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/valueobjects"
    "github.com/sophia-ecosystem/runtime-adapters/internal/ports/outbound"

// ValidateProfileSupport ensures every (capability, fault) pair in the
// profile is supported by the wrapped adapter. Decorator-native faults
// (IsDecoratorNative) skip the check — every adapter supports them. For
// adapter-synthetic faults, the wrapped adapter must implement
// ChaosCapable and report the kind in SupportedChaosFaults(cap).
//
// FaultPersistFailure on the persist boundary is validated separately
// (it never reaches an adapter). ValidateProfileSupport ignores
// profile.Persist; that's ChaosReceiptStore's concern.
//
// Returns nil if the profile is empty or absent. Returns a structured
// error listing every unsupported (capability, fault) pair.
func ValidateProfileSupport(
    registry map[valueobjects.AdapterID]outbound.Adapter,
    profile *Profile,
) error {
    if profile == nil || len(profile.Adapters) == 0 {
        return nil
    }

    var unsupported []string
    for canonical, fault := range profile.Adapters {
        if IsDecoratorNative(fault.Kind) {
            continue
        }
        // Find the adapter that owns this capability via the registry.
        adapter, cap, ok := lookupAdapterByCanonical(registry, canonical)
        if !ok {
            unsupported = append(unsupported,
                fmt.Sprintf("capability %q in profile is not registered", canonical))
            continue
        }
        cc, ok := adapter.(ChaosCapable)
        if !ok {
            unsupported = append(unsupported,
                fmt.Sprintf("capability %q (adapter %q) does not implement ChaosCapable; cannot synthesize fault %q",
                    canonical, adapter.ID(), fault.Kind))
            continue
        }
        if !containsFault(cc.SupportedChaosFaults(cap), fault.Kind) {
            unsupported = append(unsupported,
                fmt.Sprintf("capability %q (adapter %q) does not support fault %q",
                    canonical, adapter.ID(), fault.Kind))
        }
    }
    if len(unsupported) > 0 {
        return fmt.Errorf("chaos profile %q unsupported by registered adapters:\n  - %s",
            profile.Name, joinLines(unsupported))
    }
    return nil
}

// lookupAdapterByCanonical scans the registry for the adapter that owns
// the given canonical capability string. Returns the adapter, the matched
// Capability value, and ok=true on hit.
func lookupAdapterByCanonical(
    registry map[valueobjects.AdapterID]outbound.Adapter,
    canonical string,
) (outbound.Adapter, valueobjects.Capability, bool) {
    for _, a := range registry {
        for _, c := range a.Capabilities() {
            if c.Canonical() == canonical {
                return a, c, true
            }
        }
    }
    return nil, valueobjects.Capability{}, false
}

func containsFault(list []FaultKind, target FaultKind) bool {
    for _, k := range list {
        if k == target {
            return true
        }
    }
    return false
}

func joinLines(lines []string) string {
    out := lines[0]
    for _, l := range lines[1:] {
        out += "\n  - " + l
    }
    return out
}
```

```go
func (c *ChaosAdapter) dispatchFault(
    ctx context.Context,
    cap valueobjects.Capability,
    payload valueobjects.Payload,
    fault FaultConfig,
) (services.AdapterRawOutcome, error) {
    switch fault.Kind {
    case FaultLatency:
        // Sleep ctx-aware for the configured duration, then delegate.
        // Duration is taken from fault.Match["duration"] or a default; treat parse failure as bug.
        d := parseLatency(fault.Match)
        select {
        case <-ctx.Done():
            return nil, ctx.Err()
        case <-time.After(d):
        }
        return c.real.Execute(ctx, cap, payload)

    case FaultHangUntilCancel:
        <-ctx.Done()
        return nil, ctx.Err()

    case FaultInjectPanic:
        panic(fmt.Sprintf("chaos: injected panic for capability %q", cap.Canonical()))

    case FaultPersistFailure:
        // Defensive: persist faults are handled by ChaosReceiptStore, not here.
        // ValidateProfileSupport (1.4a) does not validate persist faults against
        // adapters, but a misconfigured profile could land us here. Return error.
        return nil, fmt.Errorf("chaos: persist_failure dispatched to ChaosAdapter (should be ChaosReceiptStore)")

    default:
        // Adapter-synthetic — delegate to ChaosCapable.
        cc, ok := c.real.(ChaosCapable)
        if !ok {
            return nil, fmt.Errorf("chaos: adapter %q does not implement ChaosCapable; fault %q dispatch failed (ValidateProfileSupport should have rejected this)", c.real.ID(), fault.Kind)
        }
        out, ok := cc.SyntheticOutcome(ctx, cap, payload, fault)
        if !ok {
            return nil, fmt.Errorf("chaos: adapter %q ChaosCapable did not synthesize for fault %q on capability %q", c.real.ID(), fault.Kind, cap.Canonical())
        }
        return out, nil
    }
}

func parseLatency(match map[string]string) time.Duration {
    if v, ok := match["duration"]; ok {
        if d, err := time.ParseDuration(v); err == nil {
            return d
        }
    }
    return 100 * time.Millisecond
}
```

```go
package chaos

import (
	"fmt"
	"os"

	"gopkg.in/yaml.v3"

	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/valueobjects"
)

// Catalog is the closed set of valid capability IDs (Phase 1's 8 IDs).
type Catalog interface {
	Has(c valueobjects.CapabilityID) bool
}

type rawProfile struct {
	Version     int                       `yaml:"version"`
	Name        string                    `yaml:"name"`
	Description string                    `yaml:"description"`
	Adapters    map[string]rawFault       `yaml:"adapters"`
	Persist     *rawPersistFault          `yaml:"persistence"`
	Expected    *rawExpectedOutcome       `yaml:"expected_outcome"`
}

type rawFault struct {
	Fault  string            `yaml:"fault"`
	Timing string            `yaml:"timing"`
	Match  map[string]string `yaml:"match"`
}

type rawPersistFault struct {
	Fault string `yaml:"fault"`
}

type rawExpectedOutcome struct {
	Status     string `yaml:"status"`
	ErrorClass string `yaml:"error_class"`
	RetryHint  string `yaml:"retry_hint"`
}

func parseProfile(path string, cat Catalog) (*Profile, error) {
	data, err := os.ReadFile(path)
	if err != nil {
		return nil, fmt.Errorf("read profile %q: %w", path, err)
	}
	return parseProfileBytes(data, cat)
}

func parseProfileBytes(data []byte, cat Catalog) (*Profile, error) {
	var raw rawProfile
	if err := yaml.Unmarshal(data, &raw); err != nil {
		return nil, fmt.Errorf("decode YAML: %w", err)
	}
	if raw.Version != 1 {
		return nil, fmt.Errorf("profile %q: unsupported version %d (want 1)", raw.Name, raw.Version)
	}
	if raw.Name == "" {
		return nil, fmt.Errorf("profile: name is required")
	}
	if raw.Expected == nil {
		return nil, fmt.Errorf("profile %q: expected_outcome is required (D2C3.24)", raw.Name)
	}

	out := &Profile{
		Version:     raw.Version,
		Name:        raw.Name,
		Description: raw.Description,
		Adapters:    map[valueobjects.CapabilityID]FaultConfig{},
		Expected: ExpectedOutcome{
			Status:     raw.Expected.Status,
			ErrorClass: raw.Expected.ErrorClass,
			RetryHint:  raw.Expected.RetryHint,
		},
	}
	for capStr, rf := range raw.Adapters {
		capID, err := valueobjects.ParseCapabilityID(capStr)
		if err != nil {
			return nil, fmt.Errorf("profile %q: invalid capability %q: %w", raw.Name, capStr, err)
		}
		if !cat.Has(capID) {
			return nil, fmt.Errorf("profile %q: capability %q not in Phase 1 catalog", raw.Name, capStr)
		}
		kind := FaultKind(rf.Fault)
		if !IsValidFaultKind(kind) {
			return nil, fmt.Errorf("profile %q: unknown fault kind %q for capability %q", raw.Name, rf.Fault, capStr)
		}
		out.Adapters[capID] = FaultConfig{
			Kind:   kind,
			Timing: Timing(rf.Timing),
			Match:  rf.Match,
		}
	}
	if raw.Persist != nil {
		kind := FaultKind(raw.Persist.Fault)
		if !IsValidFaultKind(kind) {
			return nil, fmt.Errorf("profile %q: unknown persist fault kind %q", raw.Name, raw.Persist.Fault)
		}
		out.Persist = &PersistFaultConfig{Kind: kind}
	}
	return out, nil
}
```

```go
package chaos

import (
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// allowlistRoots are the directory roots from which profiles may be loaded.
// (D2C3.17 / D2C3.25 / R17.) Includes both host source paths and the
// runtime-container mount path used by compose.chaos.yaml.
var allowlistRoots = []string{
	"ops/chaos/profiles/ci",
	"ops/chaos/profiles/local",
	"internal/infrastructure/chaos/testdata",
	"/etc/runtime-adapters/chaos/profiles",
}

// LoadProfile reads, validates, and parses a profile from path. Path is
// hardened: filepath.Clean → must be inside the allowlist → no parent
// traversal → no symlink escape.
func LoadProfile(path string, cat Catalog) (*Profile, error) {
	cleaned := filepath.Clean(path)
	if strings.Contains(cleaned, "..") {
		return nil, fmt.Errorf("chaos: profile path %q contains parent traversal", cleaned)
	}
	abs, err := filepath.Abs(cleaned)
	if err != nil {
		return nil, fmt.Errorf("chaos: resolve %q: %w", cleaned, err)
	}
	if !inAllowlist(abs) {
		return nil, fmt.Errorf("chaos: profile path %q not in allowlist", abs)
	}
	resolved, err := filepath.EvalSymlinks(abs)
	if err != nil {
		return nil, fmt.Errorf("chaos: eval symlinks on %q: %w", abs, err)
	}
	if !inAllowlist(resolved) {
		return nil, fmt.Errorf("chaos: profile path %q escapes allowlist via symlink", abs)
	}
	return parseProfile(resolved, cat)
}

func inAllowlist(absPath string) bool {
	for _, root := range allowlistRoots {
		var rootAbs string
		if filepath.IsAbs(root) {
			rootAbs = root
		} else {
			cwd, _ := os.Getwd()
			rootAbs = filepath.Join(cwd, root)
		}
		rel, err := filepath.Rel(rootAbs, absPath)
		if err != nil {
			continue
		}
		if !strings.HasPrefix(rel, "..") && rel != "." {
			return true
		}
	}
	return false
}
```

```go
package chaos

import (
	"errors"
	"fmt"
)

// ChaosConfigEnv mirrors the env-driven chaos config; populated by
// internal/infrastructure/config/ (extended in this task).
type ChaosConfigEnv struct {
	Enabled     bool
	ProfilePath string
}

// LoadConfig resolves the chaos config from env values + Catalog. Fails
// closed (R17): production rejects chaos; missing profile when enabled
// rejects; allowlist + path hardening rejected via LoadProfile.
func LoadConfig(env ChaosConfigEnv, runtimeEnv string, cat Catalog) (*Config, error) {
	if !env.Enabled {
		return &Config{Enabled: false, Env: runtimeEnv}, nil
	}
	if runtimeEnv == "production" {
		return nil, errors.New("chaos: refuses to start with RUNTIME_ENV=production")
	}
	if env.ProfilePath == "" {
		return nil, errors.New("chaos: RUNTIME_CHAOS_PROFILE required when enabled")
	}
	profile, err := LoadProfile(env.ProfilePath, cat)
	if err != nil {
		return nil, fmt.Errorf("chaos: %w", err)
	}
	return &Config{Enabled: true, Profile: profile, Env: runtimeEnv}, nil
}
```

```go
package chaos

import (
	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/execution/valueobjects"
	"github.com/sophia-ecosystem/runtime-adapters/internal/domain/shared"
	"github.com/sophia-ecosystem/runtime-adapters/internal/ports/outbound"
)

// MaybeWrapAdaptersWithChaos returns the adapter registry and receipt store,
// either unchanged (chaos disabled) or wrapped with ChaosAdapter and
// ChaosReceiptStore as the profile dictates. Pure function; fully testable.
//
// (D2C3.12.)
func MaybeWrapAdaptersWithChaos(
	registry map[valueobjects.AdapterID]outbound.Adapter,
	store outbound.ReceiptStore,
	cfg *Config,
	clk shared.Clock,
) (map[valueobjects.AdapterID]outbound.Adapter, outbound.ReceiptStore) {
	if cfg == nil || !cfg.Enabled {
		return registry, store
	}
	wrapped := make(map[valueobjects.AdapterID]outbound.Adapter, len(registry))
	for id, real := range registry {
		wrapped[id] = NewChaosAdapter(real, cfg.Profile, clk)
	}
	if cfg.Profile.PersistFault() != nil {
		store = NewChaosReceiptStore(store, cfg.Profile)
	}
	return wrapped, store
}
```

```go
package main

import (
	"encoding/json"
	"flag"
	"log"
	"net/http"
	"sync"
	"time"
)

type Alert struct {
	Labels   map[string]string `json:"labels"`
	Received time.Time         `json:"received"`
}

type Receiver struct {
	mu       sync.Mutex
	alerts   []Alert
	cap      int
}

func NewReceiver() *Receiver {
	return &Receiver{cap: 1000}
}

func (r *Receiver) Handler() http.Handler {
	mux := http.NewServeMux()
	mux.HandleFunc("/alerts", r.handleAlerts)
	mux.HandleFunc("/inspect", r.handleInspect)
	return mux
}

func (r *Receiver) handleAlerts(w http.ResponseWriter, req *http.Request) {
	switch req.Method {
	case http.MethodPost:
		var payload struct {
			Alerts []struct {
				Labels map[string]string `json:"labels"`
			} `json:"alerts"`
		}
		if err := json.NewDecoder(req.Body).Decode(&payload); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
		r.mu.Lock()
		now := time.Now().UTC()
		for _, a := range payload.Alerts {
			r.alerts = append(r.alerts, Alert{Labels: a.Labels, Received: now})
		}
		// Cap
		if len(r.alerts) > r.cap {
			r.alerts = r.alerts[len(r.alerts)-r.cap:]
		}
		r.mu.Unlock()
		w.WriteHeader(http.StatusOK)
	case http.MethodDelete:
		r.mu.Lock()
		r.alerts = nil
		r.mu.Unlock()
		w.WriteHeader(http.StatusOK)
	default:
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
	}
}

func (r *Receiver) handleInspect(w http.ResponseWriter, req *http.Request) {
	since := req.URL.Query().Get("since")
	r.mu.Lock()
	defer r.mu.Unlock()
	out := r.alerts
	if since != "" {
		t, err := time.Parse(time.RFC3339, since)
		if err == nil {
			filtered := make([]Alert, 0, len(out))
			for _, a := range out {
				if !a.Received.Before(t) {
					filtered = append(filtered, a)
				}
			}
			out = filtered
		}
	}
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(map[string]any{"received": out})
}

func main() {
	addr := flag.String("addr", ":8088", "listen address")
	flag.Parse()
	r := NewReceiver()
	log.Printf("receiver-stub listening on %s", *addr)
	if err := http.ListenAndServe(*addr, r.Handler()); err != nil {
		log.Fatal(err)
	}
}
```

```go
type Alert struct {
	Labels   map[string]string `json:"labels"`
	Received time.Time         `json:"received"`
}

type ReceiverClient struct {
	BaseURL string
	HTTP    *http.Client
}

type ExpectedAlert struct {
	AlertName  string
	Capability string
	Adapter    string
	Severity   string
	SLOName    string
}

func (c *ReceiverClient) WaitForAlert(
	ctx context.Context,
	expected ExpectedAlert,
	target time.Duration,
	deadline time.Duration,
) (Alert, time.Duration, bool, error) {
	start := time.Now()
	deadlineAt := start.Add(deadline)
	pollEvery := 1 * time.Second

	for {
		select {
		case <-ctx.Done():
			return Alert{}, 0, false, ctx.Err()
		default:
		}
		if time.Now().After(deadlineAt) {
			return Alert{}, time.Since(start), false,
				fmt.Errorf("canary timeout after %s waiting for %s", deadline, expected.AlertName)
		}
		alerts, err := c.listAlerts(ctx)
		if err == nil {
			for _, a := range alerts {
				if matches(a, expected) {
					latency := time.Since(start)
					breached := latency > target
					return a, latency, breached, nil
				}
			}
		}
		time.Sleep(pollEvery)
	}
}

func matches(a Alert, exp ExpectedAlert) bool {
	if exp.AlertName != "" && a.Labels["alertname"] != exp.AlertName {
		return false
	}
	if exp.Capability != "" && a.Labels["capability"] != exp.Capability {
		return false
	}
	if exp.Adapter != "" && a.Labels["adapter"] != exp.Adapter {
		return false
	}
	if exp.Severity != "" && a.Labels["severity"] != exp.Severity {
		return false
	}
	if exp.SLOName != "" && a.Labels["sloth_slo"] != exp.SLOName {
		return false
	}
	return true
}

func (c *ReceiverClient) listAlerts(ctx context.Context) ([]Alert, error) {
	req, _ := http.NewRequestWithContext(ctx, http.MethodGet, c.BaseURL+"/inspect", nil)
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	var body struct {
		Received []Alert `json:"received"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&body); err != nil {
		return nil, err
	}
	return body.Received, nil
}

func (c *ReceiverClient) AlertsMatching(exp ExpectedAlert) []Alert {
	all, _ := c.listAlerts(context.Background())
	var out []Alert
	for _, a := range all {
		if matches(a, exp) {
			out = append(out, a)
		}
	}
	return out
}

func (c *ReceiverClient) Clear(ctx context.Context) error {
	req, _ := http.NewRequestWithContext(ctx, http.MethodDelete, c.BaseURL+"/alerts", nil)
	resp, err := c.HTTP.Do(req)
	if err != nil {
		return err
	}
	resp.Body.Close()
	return nil
}
```
